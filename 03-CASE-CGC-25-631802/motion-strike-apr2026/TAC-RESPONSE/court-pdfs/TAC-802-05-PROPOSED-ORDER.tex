\section{TIER 1 --- PRIMARY DISPOSITION (Second Amended Complaint)}\label{tier-1-primary-disposition-second-amended-complaint}

\textbf{IT IS HEREBY ORDERED} that:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\item
  Plaintiff's Motion for Leave to File Second Amended Complaint (filed April 16, 2026; heard April 30, 2026), as supplemented, is \textbf{GRANTED}.
\item
  The Revised Proposed Second Amended Complaint lodged with Plaintiff's Notice of Supplemental Allegations and Errata is deemed \textbf{FILED} as of the date of this Order.
